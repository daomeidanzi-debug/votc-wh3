# game's own monthly pulse; the effect lives in our own on_action so nothing is overwritten.

# Root = Country
monthly_country_pulse = {
	on_actions = {
		votc_on_monthly_country
	}
}

votc_on_monthly_country = {
	trigger = {
		is_ai = no
	}
	effect = {
		votc_snapshot = { REASON = monthly }
		# fallback for the runner (gui/votc_runner.gui): apply queued actions
		trigger_event_silently = { id = votc_cmd.1 }
	}
}
