# The player's country is saved as scope:votc_player, so these effects also work where
# root is something else (character interactions).

votc_ensure_campaign_id = {
	if = {
		limit = { NOT = { has_global_variable = votc_campaign } }
		set_global_variable = {
			name = votc_campaign
			value = { integer_range = { min = 100000 max = 999999 } }
		}
	}
	if = {
		limit = { NOT = { has_global_variable = votc_seq } }
		set_global_variable = { name = votc_seq value = 0 }
	}
}

# THIS = the player's country. A whole snapshot (monthly, refresh command).
votc_snapshot = {
	votc_snapshot_open = { REASON = $REASON$ }
	votc_snapshot_close = yes
}

# THIS = the player's country: BEGIN, campaign data, the player and the countries that matter.
# Callers may log more (characters, a talk request) before votc_snapshot_close.
votc_snapshot_open = {
	votc_ensure_campaign_id = yes
	save_scope_as = votc_player
	save_scope_as = votc_p
	error_log = "VOTC|BEGIN|eu5|2|$REASON$"
	error_log = "VOTC|META|campaign|[GetGlobalVariable('votc_campaign').GetValue|0]"
	error_log = "VOTC|META|seq|[GetGlobalVariable('votc_seq').GetValue|0]"
	error_log = "VOTC|META|date|[GetDateString]"
	error_log = "VOTC|P|tag|[votc_p.GetTag]"
	error_log = "VOTC|P|name|[votc_p.GetNameWithNoTooltip]"
	error_log = "VOTC|P|leader|[votc_p.GetGovernment.GetRulerOrRegent.GetNameWithNoTooltip]"
	error_log = "VOTC|P|leader_title|[votc_p.GetGovernment.GetRulerTitle]"
	error_log = "VOTC|P|capital|[votc_p.GetCapital.GetNameWithNoTooltip]"
	if = {
		limit = { is_great_power = yes }
		error_log = "VOTC|P|great_power|yes"
	}
	# the player's wars and alliances are public knowledge
	every_country = {
		limit = { is_at_war_with = scope:votc_player }
		save_scope_as = votc_o
		error_log = "VOTC|P|war_with|[votc_o.GetNameWithNoTooltip]"
	}
	every_country = {
		limit = { is_allied_with = { target = scope:votc_player } }
		save_scope_as = votc_o
		error_log = "VOTC|P|allies|[votc_o.GetNameWithNoTooltip]"
	}
	every_country = {
		limit = { votc_is_relevant_country = yes }
		votc_log_country = yes
	}
}

votc_snapshot_close = {
	error_log = "VOTC|END"
}

# THIS = the country, scope:votc_player = the player's country
votc_log_country = {
	save_scope_as = votc_c
	error_log = "VOTC|C|[votc_c.GetTag]|name|[votc_c.GetNameWithNoTooltip]"
	error_log = "VOTC|C|[votc_c.GetTag]|leader|[votc_c.GetGovernment.GetRulerOrRegent.GetNameWithNoTooltip]"
	error_log = "VOTC|C|[votc_c.GetTag]|leader_title|[votc_c.GetGovernment.GetRulerTitle]"
	error_log = "VOTC|C|[votc_c.GetTag]|capital|[votc_c.GetCapital.GetNameWithNoTooltip]"
	if = {
		limit = { is_great_power = yes }
		error_log = "VOTC|C|[votc_c.GetTag]|great_power|yes"
	}
	# its wars and alliances with anyone (the world situation)
	every_country = {
		limit = { is_at_war_with = scope:votc_c }
		save_scope_as = votc_o
		error_log = "VOTC|C|[votc_c.GetTag]|war_with|[votc_o.GetNameWithNoTooltip]"
	}
	every_country = {
		limit = { is_allied_with = { target = scope:votc_c } }
		save_scope_as = votc_o
		error_log = "VOTC|C|[votc_c.GetTag]|allies|[votc_o.GetNameWithNoTooltip]"
	}
	if = {
		limit = { is_at_war_with = scope:votc_player }
		error_log = "VOTC|R|[votc_c.GetTag]|war|yes"
	}
	if = {
		limit = { is_allied_with = { target = scope:votc_player } }
		error_log = "VOTC|R|[votc_c.GetTag]|ally|yes"
	}
	if = {
		limit = { OR = { is_rival_of = scope:votc_player scope:votc_player = { is_rival_of = prev } } }
		error_log = "VOTC|R|[votc_c.GetTag]|rival|yes"
	}
	if = {
		limit = { is_subject_of = scope:votc_player }
		error_log = "VOTC|R|[votc_c.GetTag]|subject|yes"
	}
	if = {
		limit = { scope:votc_player = { is_subject_of = prev } }
		error_log = "VOTC|R|[votc_c.GetTag]|overlord|yes"
	}
	if = {
		limit = { has_truce_with = scope:votc_player }
		error_log = "VOTC|R|[votc_c.GetTag]|truce|yes"
	}
	# opinion of the player, in steps
	if = {
		limit = { opinion = { target = scope:votc_player value >= 100 } }
		error_log = "VOTC|R|[votc_c.GetTag]|opinion|80"
	}
	else_if = {
		limit = { opinion = { target = scope:votc_player value >= 30 } }
		error_log = "VOTC|R|[votc_c.GetTag]|opinion|40"
	}
	else_if = {
		limit = { opinion = { target = scope:votc_player value >= 0 } }
		error_log = "VOTC|R|[votc_c.GetTag]|opinion|10"
	}
	else_if = {
		limit = { opinion = { target = scope:votc_player value >= -30 } }
		error_log = "VOTC|R|[votc_c.GetTag]|opinion|-15"
	}
	else_if = {
		limit = { opinion = { target = scope:votc_player value >= -100 } }
		error_log = "VOTC|R|[votc_c.GetTag]|opinion|-50"
	}
	else = {
		error_log = "VOTC|R|[votc_c.GetTag]|opinion|-80"
	}
}

# THIS = the character the player chose "Talk (VOTC)" on, scope:actor = the player's country
votc_talk_to_character = {
	save_scope_as = votc_talker
	# the app's character commands (gain a trait, death...) act on this character
	set_global_variable = { name = votc_talker value = this }
	if = {
		limit = { exists = owner }
		owner = { save_scope_as = votc_talker_country }
	}
	scope:actor = { votc_snapshot_open = { REASON = talk } }
	error_log = "VOTC|META|talk|character"
	error_log = "VOTC|META|talk_tag|[votc_talker_country.GetTag]"
	error_log = "VOTC|META|talk_name|[votc_talker.GetNameWithNoTooltip]"
	votc_log_character = yes
	# the character's country, when the monthly list does not have it
	if = {
		limit = {
			exists = owner
			NOT = { owner = scope:votc_player }
		}
		owner = {
			if = {
				limit = { NOT = { votc_is_relevant_country = yes } }
				votc_log_country = yes
			}
		}
	}
	scope:actor = { votc_snapshot_close = yes }
}

# THIS = scope:votc_talker, its country saved as scope:votc_talker_country
votc_log_character = {
	error_log = "VOTC|CH|[votc_talker_country.GetTag]|[votc_talker.GetNameWithNoTooltip]|title|[votc_talker.GetRoleName]"
	error_log = "VOTC|CH|[votc_talker_country.GetTag]|[votc_talker.GetNameWithNoTooltip]|age|[votc_talker.GetAgeInYears]"
	error_log = "VOTC|CH|[votc_talker_country.GetTag]|[votc_talker.GetNameWithNoTooltip]|culture|[votc_talker.GetCulture.GetNameWithNoTooltip]"
	error_log = "VOTC|CH|[votc_talker_country.GetTag]|[votc_talker.GetNameWithNoTooltip]|religion|[votc_talker.GetReligion.GetNameWithNoTooltip]"
	error_log = "VOTC|CH|[votc_talker_country.GetTag]|[votc_talker.GetNameWithNoTooltip]|nickname|[votc_talker.GetNickname]"
	error_log = "VOTC|CH|[votc_talker_country.GetTag]|[votc_talker.GetNameWithNoTooltip]|location|[votc_talker.GetLocation.GetNameWithNoTooltip]"
	if = {
		limit = { is_female = yes }
		error_log = "VOTC|CH|[votc_talker_country.GetTag]|[votc_talker.GetNameWithNoTooltip]|gender|female"
	}
	else = {
		error_log = "VOTC|CH|[votc_talker_country.GetTag]|[votc_talker.GetNameWithNoTooltip]|gender|male"
	}
	if = {
		limit = { is_ruler = yes }
		error_log = "VOTC|CH|[votc_talker_country.GetTag]|[votc_talker.GetNameWithNoTooltip]|ruler|yes"
	}
	if = {
		limit = { is_heir = yes }
		error_log = "VOTC|CH|[votc_talker_country.GetTag]|[votc_talker.GetNameWithNoTooltip]|status|heir"
	}
	votc_log_character_abilities = yes
	votc_log_character_traits = yes
	votc_log_character_family = yes
}

# the family of scope:votc_talker: parents, spouses, children
votc_log_character_family = {
	scope:votc_talker = {
		father ?= {
			save_scope_as = votc_rel
			error_log = "VOTC|CH|[votc_talker_country.GetTag]|[votc_talker.GetNameWithNoTooltip]|father|[votc_rel.GetNameWithNoTooltip]"
		}
		mother ?= {
			save_scope_as = votc_rel
			error_log = "VOTC|CH|[votc_talker_country.GetTag]|[votc_talker.GetNameWithNoTooltip]|mother|[votc_rel.GetNameWithNoTooltip]"
		}
		every_spouse = {
			save_scope_as = votc_rel
			error_log = "VOTC|CH|[votc_talker_country.GetTag]|[votc_talker.GetNameWithNoTooltip]|spouse|[votc_rel.GetNameWithNoTooltip]"
		}
		every_child = {
			save_scope_as = votc_rel
			error_log = "VOTC|CH|[votc_talker_country.GetTag]|[votc_talker.GetNameWithNoTooltip]|children|[votc_rel.GetNameWithNoTooltip]"
		}
	}
}

# administrative / diplomatic / military ability, in steps of the 0-100 scale
votc_log_character_abilities = {
	votc_log_ability = { A = adm }
	votc_log_ability = { A = dip }
	votc_log_ability = { A = mil }
}

votc_log_ability = {
	if = {
		limit = { $A$ >= 80 }
		error_log = "VOTC|CH|[votc_talker_country.GetTag]|[votc_talker.GetNameWithNoTooltip]|$A$_ability|80+ (excellent)"
	}
	else_if = {
		limit = { $A$ >= 60 }
		error_log = "VOTC|CH|[votc_talker_country.GetTag]|[votc_talker.GetNameWithNoTooltip]|$A$_ability|60-79 (good)"
	}
	else_if = {
		limit = { $A$ >= 40 }
		error_log = "VOTC|CH|[votc_talker_country.GetTag]|[votc_talker.GetNameWithNoTooltip]|$A$_ability|40-59 (average)"
	}
	else_if = {
		limit = { $A$ >= 20 }
		error_log = "VOTC|CH|[votc_talker_country.GetTag]|[votc_talker.GetNameWithNoTooltip]|$A$_ability|20-39 (poor)"
	}
	else = {
		error_log = "VOTC|CH|[votc_talker_country.GetTag]|[votc_talker.GetNameWithNoTooltip]|$A$_ability|under 20 (terrible)"
	}
}
