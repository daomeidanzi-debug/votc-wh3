# VOTC Bridge (GPL-3.0, see LICENSE.txt): log a snapshot of the player's
# country and the countries that matter to it, so the VOTC app always has fresh data.
on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = { is_ai = no }
				votc_snapshot = yes
			}
		}
	}
	on_monthly = {
		effect = {
			if = {
				limit = { is_ai = no }
				votc_snapshot = yes
			}
		}
	}
	# apply the actions the VOTC app queued (events/votc_commands.txt, rewritten by the app and
	# reloaded by the file watcher of debug mode); the event does nothing once everything has run
	on_daily = {
		effect = {
			if = {
				limit = { is_ai = no }
				country_event = votc_cmd.1
			}
		}
	}
}
